Module app.routes

HTTP routes for SmartBank (server-rendered + a small JSON API).

All routes live on a single Flask :class:~flask.Blueprint called bank. Route handlers are thin — they call into :mod:app.models for every piece of business logic and never issue SQL directly. Two view decorators (:func:login_required(), :func:admin_required()) implement the authorization layer declaratively.

def account_detail(account_number):
    db = get_db()
    acc = m.get_account(db, account_number)
    if acc is None or acc["user_id"] != session["user_id"]:
        abort(404)
    history = m.transaction_history(db, account_number)
    return render_template("account.html", account=acc, history=history)

def admin_required(view):
    """View decorator — ``403 Forbidden`` unless ``session.role == 'admin'``."""
    @wraps(view)
    def wrapped(*a, **kw):
        if session.get("role") != "admin":
            abort(403)
        return view(*a, **kw)
    return wrapped

def login()

@bp.route("/login", methods=["GET", "POST"])
def login():
    if request.method == "POST":
        try:
            user = m.authenticate(
                get_db(),
                request.form["username"].strip(),
                request.form["password"],
            )
            session.clear()
            session["user_id"] = user["user_id"]
            session["username"] = user["username"]
            session["role"] = user["role"]
            if user["role"] == "admin":
                return redirect(url_for("bank.admin_panel"))
            return redirect(url_for("bank.dashboard"))
        except m.AuthError as e:
            flash(str(e), "error")
    return render_template("login.html")

def login_required(view):
    """View decorator — redirect to ``/login`` if no session is active."""
    @wraps(view)
    def wrapped(*a, **kw):
        if "user_id" not in session:
            return redirect(url_for("bank.login"))
        return view(*a, **kw)
    return wrapped

@bp.route("/profile", methods=["GET", "POST"])
@login_required
def profile():
    db = get_db()
    if request.method == "POST":
        m.update_profile(
            db, session["user_id"],
            full_name=request.form.get("full_name") or None,
            address=request.form.get("address") or None,
            phone=request.form.get("phone") or None,
            email=request.form.get("email") or None,
        )
        flash("Profile updated", "success")
        return redirect(url_for("bank.profile"))
    user = m.get_user(db, session["user_id"])
    return render_template("profile.html", user=user)
